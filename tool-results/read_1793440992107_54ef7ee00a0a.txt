   302→function OrdersManagement() {

   305→  const [loading, setLoading] = useState(true);
   306→  const [savingId, setSavingId] = useState<string | null>(null);

   309→  const loadOrders = useCallback(() => {

   311→      .then((r) => (r.ok ? r.json() : { orders: [] }))
   312→      .then((data) => setOrders(data.orders || []))
   313→      .finally(() => setLoading(false));
   314→  }, []);
   315→
   316→  useEffect(() => {
   317→    loadOrders();
   318→  }, [loadOrders]);
   319→
   320→  const patchOrder = async (orderNumber: string, body: Record<string, string>) => {
   321→    setSavingId(orderNumber);
   322→    try {
   323→      const res = await fetch(`/api/orders/${orderNumber}`, {
   324→        method: 'PATCH',
   325→        headers: { 'Content-Type': 'application/json' },
   326→        body: JSON.stringify(body),
   327→      });
   328→      const data = await res.json();
   329→      if (!res.ok) throw new Error(data.error || 'Update failed');
   330→      setOrders((list) => list.map((o) => (o.order_number === orderNumber ? data.order : o)));
   331→      dispatch({ type: 'SET_NOTIFICATION', notification: { type: 'success', message: `Order ${orderNumber} updated` } });
   332→    } catch (error) {
   333→      dispatch({ type: 'SET_NOTIFICATION', notification: { type: 'error', message: (error as Error).message } });
   334→    } finally {
   335→      setSavingId(null);
   336→    }
   337→  };
   338→
   339→  const visible = filter === 'all' ? orders : orders.filter((o) => o.status === filter);
   340→
   341→  return (
   342→    <div className="space-y-4">

   504→function CustomersManagement() {
   505→  const { state, dispatch } = useApp();
   506→  const [users, setUsers] = useState<AdminUserRow[]>([]);
   507→  const [loading, setLoading] = useState(true);
   508→  const [savingId, setSavingId] = useState<string | null>(null);
   509→  const isSuperAdmin = state.user?.role === 'super_admin';
   510→
   511→  const loadUsers = useCallback(() => {
   512→    return fetch('/api/admin/users')
   513→      .then((r) => (r.ok ? r.json() : { users: [] }))
   514→      .then((data) => setUsers(data.users || []))
   515→      .finally(() => setLoading(false));
   516→  }, []);
   517→
   518→  useEffect(() => {
   519→    loadUsers();
   520→  }, [loadUsers]);
   521→
   522→  const patchUser = async (id: string, body: Record<string, string | boolean>, okMessage: string) => {
   523→    setSavingId(id);
   524→    try {
   525→      const res = await fetch(`/api/admin/users/${id}`, {
   526→        method: 'PATCH',
   527→        headers: { 'Content-Type': 'application/json' },
   528→        body: JSON.stringify(body),
   529→      });
   530→      const data = await res.json();
   531→      if (!res.ok) throw new Error(data.error || 'Update failed');
   532→      setUsers((list) => list.map((u) => (u.id === id ? data.user : u)));
   533→      dispatch({ type: 'SET_NOTIFICATION', notification: { type: 'success', message: okMessage } });
   534→    } catch (error) {
   535→      dispatch({ type: 'SET_NOTIFICATION', notification: { type: 'error', message: (error as Error).message } });
   536→    } finally {
   537→      setSavingId(null);
   538→    }
   539→  };
   540→
   541→  return (
   542→    <div className="bg-white border border-gray-200 rounded-xl overflow-hidden">
   543→      <div className="overflow-x-auto">
   544→        <table className="w-full text-sm">
   545→          <thead className="bg-gray-50">
   546→            <tr>
   547→              <th className="text-left px-4 py-3 font-medium text-navy-600">Customer</th>
   548→              <th className="text-left px-4 py-3 font-medium text-navy-600">Email</th>
   549→              <th className="text-center px-4 py-3 font-medium text-navy-600">Role</th>
   550→              <th className="text-right px-4 py-3 font-medium text-navy-600">Orders</th>
   551→              <th className="text-center px-4 py-3 font-medium text-navy-600">Status</th>
   552→              <th className="text-center px-4 py-3 font-medium text-navy-600">Actions</th>
   553→            </tr>
   554→          </thead>
   555→          <tbody>
   556→            {users.map((u) => {
   557→              const self = u.id === state.user?.id;
   558→              const manageable = u.role === 'customer' || isSuperAdmin;
   559→              return (
   560→                <tr key={u.id} className="border-t border-gray-100 hover:bg-gray-50">
   561→                  <td className="px-4 py-3"><div className="flex items-center gap-2"><div className="w-8 h-8 bg-primary-100 rounded-full flex items-center justify-center text-xs font-medium text-primary-700">{u.name?.split(' ').map((n) => n[0]).join('')}</div><span className="font-medium text-navy-900">{u.name}{self && <span className="text-[10px] text-navy-400 ml-1">(you)</span>}</span></div></td>
   562→                  <td className="px-4 py-3 text-navy-600">{u.email}</td>
   563→                  <td className="px-4 py-3 text-center">
   564→                    {isSuperAdmin && !self ? (
   565→                      <select
   566→                        value={u.role}
   567→                        disabled={savingId === u.id}
   568→                        onChange={(e) => patchUser(u.id, { role: e.target.value }, `${u.name} is now ${e.target.value.replace('_', ' ')}`)}
   569→                        className="text-xs border border-gray-200 rounded-lg px-2 py-1.5 bg-white focus:outline-none focus:ring-2 focus:ring-primary-500 disabled:opacity-50"
   570→                        aria-label={`Change role for ${u.email}`}
   571→                      >
   572→                        <option value="customer">customer</option>
   573→                        <option value="admin">admin</option>
   574→                        <option value="super_admin">super_admin</option>
   575→                      </select>
   576→                    ) : (
   577→                      <span className={`text-xs px-2 py-1 rounded-full ${u.role === 'super_admin' ? 'bg-purple-100 text-purple-700' : u.role === 'admin' ? 'bg-blue-100 text-blue-700' : 'bg-gray-100 text-gray-700'}`}>{u.role}</span>
   578→                    )}
   579→                  </td>
   580→                  <td className="px-4 py-3 text-right text-navy-700">{u.order_count}</td>
   581→                  <td className="px-4 py-3 text-center"><span className={`text-xs px-2 py-1 rounded-full ${u.is_active ? 'bg-green-100 text-green-700' : 'bg-red-100 text-red-700'}`}>{u.is_active ? 'Active' : 'Disabled'}</span></td>
   582→                  <td className="px-4 py-3 text-center">
