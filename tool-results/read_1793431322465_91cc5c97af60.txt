   149→function ProductsManagement() {
   150→  const { state } = useApp();
   151→  const { saveProduct, deleteProduct } = useAdminMutations();
   152→  const [searchTerm, setSearchTerm] = useState('');
   153→  const [showAddForm, setShowAddForm] = useState(false);
   154→  const [editProduct, setEditProduct] = useState<Product | null>(null);
   155→  const [categories, setCategories] = useState<Category[]>([]);
   156→  const [formData, setFormData] = useState(emptyForm);
   157→
   158→  useEffect(() => {
   159→    fetch('/api/categories').then((r) => r.json()).then((data) => setCategories(data.categories || []));
   160→  }, []);
   161→
   162→  const filtered = state.products.filter((p) => p.name.toLowerCase().includes(searchTerm.toLowerCase()));
   163→
   164→  const handleSave = async () => {
   165→    if (!formData.name || !formData.price) return;
   166→    const ok = await saveProduct(formData, editProduct);
   167→    if (ok) {
   168→      setShowAddForm(false);
   169→      setEditProduct(null);
   170→      setFormData(emptyForm);
   171→    }
   172→  };
   173→
   174→  const handleDelete = async (productId: string) => {
   175→    if (confirm('Are you sure you want to delete this product?')) {
   176→      await deleteProduct(productId);
   177→    }
   178→  };
   179→
   180→  const startEdit = (product: Product) => {
   181→    setEditProduct(product);
   182→    setFormData({
   183→      name: product.name,
   184→      slug: product.slug,
   185→      price: String(product.price),
   186→      compare_price: product.compare_price ? String(product.compare_price) : '',
   187→      brand: product.brand || '',
   188→      sku: product.sku || '',
   189→      stock: String(product.stock),
   190→      description: product.description || '',
   191→      image_url: product.images?.[0] || '',
   192→      category_id: product.category_id || '',
   193→    });
   194→    setShowAddForm(true);
   195→  };
   196→
   197→  return (
   198→    <div className="space-y-4">
   199→      <div className="flex flex-col sm:flex-row justify-between items-start sm:items-center gap-3">
   200→        <div className="relative flex-1 max-w-md">
   201→          <Search size={16} className="absolute left-3 top-1/2 -translate-y-1/2 text-navy-400" />
   202→          <input type="text" value={searchTerm} onChange={(e) => setSearchTerm(e.target.value)} placeholder="Search products..." className="w-full pl-10 pr-4 py-2 border border-gray-200 rounded-lg text-sm focus:outline-none focus:ring-2 focus:ring-primary-500" />
   203→        </div>
   204→        <button onClick={() => { setShowAddForm(true); setEditProduct(null); setFormData(emptyForm); }} className="flex items-center gap-2 px-4 py-2 bg-primary-600 text-white rounded-lg text-sm font-medium hover:bg-primary-700"><Plus size={14} /> Add Product</button>
   205→      </div>
   206→
   207→      {showAddForm && (
   208→        <div className="bg-white border border-gray-200 rounded-xl p-6 animate-fade-in">
   209→          <h3 className="font-bold text-navy-900 mb-4">{editProduct ? 'Edit Product' : 'Add New Product'}</h3>
   210→          <div className="grid sm:grid-cols-2 gap-4">
   211→            <div><label className="block text-sm font-medium text-navy-700 mb-1">Name *</label><input type="text" value={formData.name} onChange={(e) => setFormData({ ...formData, name: e.target.value })} className="w-full px-3 py-2 border border-gray-200 rounded-lg text-sm focus:outline-none focus:ring-2 focus:ring-primary-500" /></div>
   212→            <div><label className="block text-sm font-medium text-navy-700 mb-1">Slug</label><input type="text" value={formData.slug} onChange={(e) => setFormData({ ...formData, slug: e.target.value })} className="w-full px-3 py-2 border border-gray-200 rounded-lg text-sm focus:outline-none focus:ring-2 focus:ring-primary-500" placeholder="auto-generated if empty" /></div>
   213→            <div><label className="block text-sm font-medium text-navy-700 mb-1">Price (₹) *</label><input type="number" value={formData.price} onChange={(e) => setFormData({ ...formData, price: e.target.value })} className="w-full px-3 py-2 border border-gray-200 rounded-lg text-sm focus:outline-none focus:ring-2 focus:ring-primary-500" /></div>

   327→  const [formData, setFormData] = useState({ title: '', subtitle: '', cta_text: '', cta_link: '', image_url: '', background_color: 'from-indigo-600 to-purple-600' });
   328→
   329→  const handleAdd = async () => {
   330→    if (!formData.title) return;
   331→    const ok = await addBanner(formData);
   332→    if (ok) {
   333→      setShowAddForm(false);
   334→      setFormData({ title: '', subtitle: '', cta_text: '', cta_link: '', image_url: '', background_color: 'from-indigo-600 to-purple-600' });
   335→    }
   336→  };
   337→
   338→  const handleDelete = async (bannerId: string) => {
   339→    if (confirm('Delete this banner?')) {
   340→      await deleteBanner(bannerId);
   341→    }
   342→  };
   343→
   344→  return (
   345→    <div className="space-y-4">
   346→      <div className="flex justify-between items-center">
   347→        <h2 className="text-lg font-bold text-navy-900">Banners ({state.banners.length})</h2>
   348→        <button onClick={() => setShowAddForm(!showAddForm)} className="flex items-center gap-2 px-4 py-2 bg-primary-600 text-white rounded-lg text-sm font-medium hover:bg-primary-700"><Plus size={14} /> Add Banner</button>
   349→      </div>
   350→
   351→      {showAddForm && (
   352→        <div className="bg-white border border-gray-200 rounded-xl p-6 animate-fade-in">
   353→          <h3 className="font-bold text-navy-900 mb-4">Create Banner</h3>
   354→          <div className="grid sm:grid-cols-2 gap-4">
   355→            <div><label className="block text-sm font-medium text-navy-700 mb-1">Title *</label><input type="text" value={formData.title} onChange={(e) => setFormData({ ...formData, title: e.target.value })} className="w-full px-3 py-2 border border-gray-200 rounded-lg text-sm focus:outline-none focus:ring-2 focus:ring-primary-500" /></div>
   356→            <div><label className="block text-sm font-medium text-navy-700 mb-1">Subtitle</label><input type="text" value={formData.subtitle} onChange={(e) => setFormData({ ...formData, subtitle: e.target.value })} className="w-full px-3 py-2 border border-gray-200 rounded-lg text-sm focus:outline-none focus:ring-2 focus:ring-primary-500" /></div>
   357→            <div><label className="block text-sm font-medium text-navy-700 mb-1">CTA Text</label><input type="text" value={formData.cta_text} onChange={(e) => setFormData({ ...formData, cta_text: e.target.value })} className="w-full px-3 py-2 border border-gray-200 rounded-lg text-sm focus:outline-none focus:ring-2 focus:ring-primary-500" /></div>
   358→            <div><label className="block text-sm font-medium text-navy-700 mb-1">CTA Link</label><input type="text" value={formData.cta_link} onChange={(e) => setFormData({ ...formData, cta_link: e.target.value })} className="w-full px-3 py-2 border border-gray-200 rounded-lg text-sm focus:outline-none focus:ring-2 focus:ring-primary-500" /></div>
   359→            <div className="sm:col-span-2"><label className="block text-sm font-medium text-navy-700 mb-1">Image URL</label><input type="text" value={formData.image_url} onChange={(e) => setFormData({ ...formData, image_url: e.target.value })} className="w-full px-3 py-2 border border-gray-200 rounded-lg text-sm focus:outline-none focus:ring-2 focus:ring-primary-500" /></div>
   360→            <div><label className="block text-sm font-medium text-navy-700 mb-1">Background Gradient</label>
   361→              <select value={formData.background_color} onChange={(e) => setFormData({ ...formData, background_color: e.target.value })} className="w-full px-3 py-2 border border-gray-200 rounded-lg text-sm focus:outline-none focus:ring-2 focus:ring-primary-500">
   362→                <option value="from-indigo-600 to-purple-600">Indigo → Purple</option>
   363→                <option value="from-emerald-600 to-teal-600">Emerald → Teal</option>
   364→                <option value="from-amber-500 to-orange-600">Amber → Orange</option>
   365→                <option value="from-rose-500 to-pink-600">Rose → Pink</option>
   366→                <option value="from-navy-800 to-navy-900">Navy Dark</option>
   367→              </select>
   368→            </div>
   369→          </div>
   370→          <div className="flex gap-3 mt-4">
   371→            <button onClick={handleAdd} className="px-6 py-2 bg-primary-600 text-white rounded-lg text-sm font-medium hover:bg-primary-700">Create Banner</button>
   372→            <button onClick={() => setShowAddForm(false)} className="px-6 py-2 border border-gray-200 rounded-lg text-sm font-medium text-navy-700 hover:bg-gray-50">Cancel</button>
   373→          </div>
   374→        </div>
   375→      )}
   376→
   377→      <div className="grid gap-4">
   378→        {state.banners.map((banner: Banner) => (
   379→          <div key={banner.id} className={`bg-gradient-to-r ${banner.background_color || 'from-indigo-600 to-purple-600'} rounded-xl p-5 text-white relative overflow-hidden`}>
   380→            {banner.image_url && <img src={banner.image_url} alt="" className="absolute inset-0 w-full h-full object-cover opacity-20" />}
   381→            <div className="relative flex justify-between items-center">
   382→              <div>
   383→                <h3 className="text-xl font-bold">{banner.title}</h3>
   384→                <p className="opacity-90">{banner.subtitle}</p>
   385→                {banner.cta_text && <span className="inline-block mt-2 px-3 py-1 bg-white/20 rounded text-sm">{banner.cta_text}</span>}
   386→              </div>
   387→              <div className="flex gap-2">
   388→                <button onClick={() => toggleBanner(banner)} className={`px-3 py-1 rounded text-xs font-medium ${banner.is_active ? 'bg-green-500/30' : 'bg-red-500/30'}`}>{banner.is_active ? 'Active' : 'Inactive'}</button>
   389→                <button onClick={() => handleDelete(banner.id)} aria-label="Delete banner" className="p-1.5 bg-white/20 rounded hover:bg-white/30"><Trash2 size={14} /></button>
   390→              </div>
   391→            </div>
   392→          </div>
   393→        ))}
   394→      </div>
   395→    </div>
   396→  );
   397→}

   399→function CustomersManagement() {

   439→      </div>
