   208→      setEditProduct(null);

   348→function BannersManagement() {
   349→  const { state, dispatch } = useApp();
   350→  const [showAddForm, setShowAddForm] = useState(false);
   351→  const [formData, setFormData] = useState({ title: '', subtitle: '', cta_text: '', cta_link: '', image_url: '', background_color: 'from-indigo-600 to-purple-600' });
   352→
   353→  const handleAdd = async () => {
   354→    if (!formData.title) return;
   355→    try {
   356→      const { db } = await import('../lib/db');
   357→      const result = await db`
   358→        INSERT INTO banners (title, subtitle, cta_text, cta_link, image_url, background_color, sort_order, is_active)
   359→        VALUES (${formData.title}, ${formData.subtitle}, ${formData.cta_text}, ${formData.cta_link}, ${formData.image_url}, ${formData.background_color}, ${state.banners.length + 1}, true)
   360→        RETURNING *
   361→      `;
   362→      const newBanner: Banner = {
   363→        id: result[0].id,
   364→        title: result[0].title,
   365→        subtitle: result[0].subtitle,
   366→        cta_text: result[0].cta_text,
   367→        cta_link: result[0].cta_link,
   368→        image_url: result[0].image_url,
   369→        background_color: result[0].background_color,
   370→        is_active: result[0].is_active,
   371→        sort_order: result[0].sort_order,
   372→      };
   373→      dispatch({ type: 'ADD_BANNER', banner: newBanner });
   374→      dispatch({ type: 'SET_NOTIFICATION', notification: { type: 'success', message: 'Banner created in database!' } });
   375→      setShowAddForm(false);
   376→      setFormData({ title: '', subtitle: '', cta_text: '', cta_link: '', image_url: '', background_color: 'from-indigo-600 to-purple-600' });
   377→    } catch (error: any) {
   378→      dispatch({ type: 'SET_NOTIFICATION', notification: { type: 'error', message: 'Failed to create banner: ' + error.message } });
   379→    }
   380→  };
   381→
   382→  const handleDelete = async (bannerId: string) => {
   383→    if (confirm('Delete this banner?')) {
   384→      try {
   385→        const { db } = await import('../lib/db');
   386→        await db`DELETE FROM banners WHERE id = ${bannerId}`;
   387→        dispatch({ type: 'DELETE_BANNER', bannerId });
   388→        dispatch({ type: 'SET_NOTIFICATION', notification: { type: 'success', message: 'Banner deleted from database!' } });
   389→      } catch (error: any) {
   390→        dispatch({ type: 'SET_NOTIFICATION', notification: { type: 'error', message: 'Failed to delete banner: ' + error.message } });
   391→      }
   392→    }
   393→  };
   394→
   395→  const toggleActive = async (banner: any) => {
   396→    try {
   397→      const { db } = await import('../lib/db');
   398→      const newActive = !banner.is_active;
   399→      await db`UPDATE banners SET is_active = ${newActive} WHERE id = ${banner.id}`;
   400→      const updated = { ...banner, is_active: newActive };
   401→      dispatch({ type: 'UPDATE_BANNER', banner: updated });
   402→    } catch (error: any) {
   403→      dispatch({ type: 'SET_NOTIFICATION', notification: { type: 'error', message: 'Failed to update banner: ' + error.message } });
   404→    }
   405→  };
   406→
   407→  return (
   408→    <div className="space-y-4">
   409→      <div className="flex justify-between items-center">
   410→        <h2 className="text-lg font-bold text-navy-900">Banners ({state.banners.length})</h2>
   411→        <button onClick={() => setShowAddForm(!showAddForm)} className="flex items-center gap-2 px-4 py-2 bg-primary-600 text-white rounded-lg text-sm font-medium hover:bg-primary-700"><Plus size={14} /> Add Banner</button>
   412→      </div>
   413→
   414→      {showAddForm && (
   415→        <div className="bg-white border border-gray-200 rounded-xl p-6 animate-fade-in">
   416→          <h3 className="font-bold text-navy-900 mb-4">Create Banner</h3>
   417→          <div className="grid sm:grid-cols-2 gap-4">
   418→            <div><label className="block text-sm font-medium text-navy-700 mb-1">Title *</label><input type="text" value={formData.title} onChange={(e) => setFormData({ ...formData, title: e.target.value })} className="w-full px-3 py-2 border border-gray-200 rounded-lg text-sm focus:outline-none focus:ring-2 focus:ring-primary-500" /></div>
   419→            <div><label className="block text-sm font-medium text-navy-700 mb-1">Subtitle</label><input type="text" value={formData.subtitle} onChange={(e) => setFormData({ ...formData, subtitle: e.target.value })} className="w-full px-3 py-2 border border-gray-200 rounded-lg text-sm focus:outline-none focus:ring-2 focus:ring-primary-500" /></div>
   420→            <div><label className="block text-sm font-medium text-navy-700 mb-1">CTA Text</label><input type="text" value={formData.cta_text} onChange={(e) => setFormData({ ...formData, cta_text: e.target.value })} className="w-full px-3 py-2 border border-gray-200 rounded-lg text-sm focus:outline-none focus:ring-2 focus:ring-primary-500" /></div>
   421→            <div><label className="block text-sm font-medium text-navy-700 mb-1">CTA Link</label><input type="text" value={formData.cta_link} onChange={(e) => setFormData({ ...formData, cta_link: e.target.value })} className="w-full px-3 py-2 border border-gray-200 rounded-lg text-sm focus:outline-none focus:ring-2 focus:ring-primary-500" /></div>
   422→            <div className="sm:col-span-2"><label className="block text-sm font-medium text-navy-700 mb-1">Image URL</label><input type="text" value={formData.image_url} onChange={(e) => setFormData({ ...formData, image_url: e.target.value })} className="w-full px-3 py-2 border border-gray-200 rounded-lg text-sm focus:outline-none focus:ring-2 focus:ring-primary-500" /></div>
   423→            <div><label className="block text-sm font-medium text-navy-700 mb-1">Background Gradient</label>
   424→              <select value={formData.background_color} onChange={(e) => setFormData({ ...formData, background_color: e.target.value })} className="w-full px-3 py-2 border border-gray-200 rounded-lg text-sm focus:outline-none focus:ring-2 focus:ring-primary-500">
   425→                <option value="from-indigo-600 to-purple-600">Indigo → Purple</option>
   426→                <option value="from-emerald-600 to-teal-600">Emerald → Teal</option>
   427→                <option value="from-amber-500 to-orange-600">Amber → Orange</option>
   428→                <option value="from-rose-500 to-pink-600">Rose → Pink</option>
   429→                <option value="from-navy-800 to-navy-900">Navy Dark</option>
   430→              </select>
   431→            </div>
   432→          </div>
   433→          <div className="flex gap-3 mt-4">
   434→            <button onClick={handleAdd} className="px-6 py-2 bg-primary-600 text-white rounded-lg text-sm font-medium hover:bg-primary-700">Create Banner</button>
   435→            <button onClick={() => setShowAddForm(false)} className="px-6 py-2 border border-gray-200 rounded-lg text-sm font-medium text-navy-700 hover:bg-gray-50">Cancel</button>
   436→          </div>
   437→        </div>
   438→      )}
   439→
   440→      <div className="grid gap-4">
   441→        {state.banners.map(banner => (
   442→          <div key={banner.id} className={`bg-gradient-to-r ${banner.background_color || 'from-indigo-600 to-purple-600'} rounded-xl p-5 text-white relative overflow-hidden`}>
   443→            {banner.image_url && <img src={banner.image_url} alt="" className="absolute inset-0 w-full h-full object-cover opacity-20" />}
   444→            <div className="relative flex justify-between items-center">
   445→              <div>
   446→                <h3 className="text-xl font-bold">{banner.title}</h3>
   447→                <p className="opacity-90">{banner.subtitle}</p>

   451→                <button onClick={() => toggleActive(banner)} className={`px-3 py-1 rounded text-xs font-medium ${banner.is_active ? 'bg-green-500/30' : 'bg-red-500/30'}`}>{banner.is_active ? 'Active' : 'Inactive'}</button>
   452→                <button onClick={() => handleDelete(banner.id)} className="p-1.5 bg-white/20 rounded hover:bg-white/30"><Trash2 size={14} /></button>
   453→              </div>
   454→            </div>
   455→          </div>
